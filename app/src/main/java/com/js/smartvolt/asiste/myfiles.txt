package com.js.smartvolt.asiste

import android.os.Bundle
import androidx.compose.material3.Surface
import androidx.activity.compose.BackHandler
import androidx.activity.compose.setContent
import androidx.activity.enableEdgeToEdge
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.collectAsState
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.fragment.app.FragmentActivity
import androidx.lifecycle.viewmodel.compose.viewModel
import com.js.smartvolt.asiste.model.data.local.DatabaseProvider
import com.js.smartvolt.asiste.ui.theme.AsisteSVTheme
import com.js.smartvolt.asiste.ui.view.LoginView
import com.js.smartvolt.asiste.ui.view.MainView
import com.js.smartvolt.asiste.ui.view.admin.AdminMainView
import com.js.smartvolt.asiste.ui.view.capataz.CapatazMainView
import com.js.smartvolt.asiste.viewmodel.OlvideClaveViewModel
import com.js.smartvolt.asiste.viewmodel.OlvideClaveViewModelFactory
import com.js.smartvolt.asiste.ui.view.OlvideContrasenaView
import androidx.compose.runtime.getValue
import androidx.compose.runtime.setValue
import com.js.smartvolt.asiste.viewmodel.LoginEstado
import com.js.smartvolt.asiste.viewmodel.LoginViewModel
import com.js.smartvolt.asiste.viewmodel.LoginViewModelFactory
import com.js.smartvolt.asiste.session.SessionManager
import com.js.smartvolt.asiste.data.repository.AuthRepository


class MainActivity : FragmentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()

        setContent {
            AsisteSVTheme() {
                // ---- Navegación: "login" | "olvide" | "inicio" ----
                var pantalla by remember { mutableStateOf("login") }
                var esAdmin by remember { mutableStateOf(false) }
                var codigoOperario by remember { mutableStateOf<String?>(null) }

                val context = LocalContext.current
                val usuarioDao = remember { DatabaseProvider.getUsuarioDao(context) }
                val vm: LoginViewModel = viewModel(factory = LoginViewModelFactory(AuthRepository(usuarioDao)))
                val estado by vm.estado.collectAsState()


                // Si el login fue exitoso -> pantalla de inicio
                LaunchedEffect(estado) {
                    if (estado is LoginEstado.Exito) {
                        codigoOperario = (estado as LoginEstado.Exito).codigo
                        val rol = SessionManager.rol
                        pantalla = when (rol) {
                            com.js.smartvolt.asiste.model.admin.Rol.ADMIN -> "admin"
                            com.js.smartvolt.asiste.model.admin.Rol.CAPATAZ -> "capataz"
                            else -> "inicio"
                        }
                    }
                }

                // Botón atrás: desde cualquier pantalla (excepto admin, que gestiona su propio back) vuelve al login
                BackHandler (enabled = pantalla != "login" && pantalla != "admin") {
                    vm.limpiarEstado()
                    pantalla = "login"
                }

                when (pantalla) {
                    "login" -> Surface(modifier = Modifier.fillMaxSize()) {
                        LoginView(
                            onLogin = { correo, pass -> esAdmin = correo.trim().lowercase().contains("admin"); vm.login(correo, pass) },
                            onOlvideClave = { pantalla = "olvide" },
                            cargando = estado is LoginEstado.Cargando,
                            error = (estado as? LoginEstado.Error)?.mensaje
                        )
                    }

                    "olvide" -> {
                        val vmOlvide: OlvideClaveViewModel = viewModel(
                            factory = OlvideClaveViewModelFactory(usuarioDao)
                        )
                        val estadoOlvide by vmOlvide.estado.collectAsState()

                        // Al entrar a la pantalla se limpia el estado anterior
                        LaunchedEffect(Unit) { vmOlvide.reset() }

                        // OJO: aquí va el composable de la pantalla, no el ViewModel
                        OlvideContrasenaView(
                            onVolver = {
                                vmOlvide.reset()
                                vm.limpiarEstado()
                                pantalla = "login"
                            },
                            estado = estadoOlvide,
                            onEnviar = { vmOlvide.enviar(it) }
                        )
                    }

                    "inicio" -> MainView(codigoOperario = codigoOperario ?: "OP-014", onCerrarSesion = { vm.limpiarEstado(); pantalla = "login" })
                    "admin" -> AdminMainView(onCerrarSesion = { vm.limpiarEstado(); pantalla = "login" })
                    "capataz" -> CapatazMainView(onCerrarSesion = { vm.limpiarEstado(); pantalla = "login" })
                }
            }
        }
    }
}

package com.js.smartvolt.asiste.data.location

import android.Manifest
import android.annotation.SuppressLint
import android.content.Context
import android.content.pm.PackageManager
import android.location.LocationManager
import android.os.Build
import androidx.core.content.ContextCompat
import androidx.core.location.LocationManagerCompat
import com.google.android.gms.location.LocationServices
import com.google.android.gms.location.Priority
import com.google.android.gms.tasks.CancellationTokenSource
import kotlinx.coroutines.tasks.await
import kotlinx.coroutines.withTimeoutOrNull
import kotlin.time.Duration.Companion.milliseconds

data class LecturaUbicacion(
    val latitud: Double,
    val longitud: Double,
    val precisionMetros: Float,
    val simulada: Boolean
)

sealed interface ResultadoUbicacion {
    data class Ok(val lectura: LecturaUbicacion) : ResultadoUbicacion
    data object SinPermiso : ResultadoUbicacion
    data object GpsApagado : ResultadoUbicacion
    data class Error(val mensaje: String) : ResultadoUbicacion
}

class UbicacionProvider(context: Context) {

    private val appContext = context.applicationContext
    private val client = LocationServices.getFusedLocationProviderClient(appContext)

    // Solo cuenta el permiso PRECISO
    private fun tienePermiso(): Boolean =
        ContextCompat.checkSelfPermission(appContext, Manifest.permission.ACCESS_FINE_LOCATION) ==
                PackageManager.PERMISSION_GRANTED

    private fun gpsActivo(): Boolean {
        val lm = appContext.getSystemService(Context.LOCATION_SERVICE) as LocationManager
        return LocationManagerCompat.isLocationEnabled(lm)
    }

    @SuppressLint("MissingPermission")
    suspend fun obtener(): ResultadoUbicacion {
        if (!tienePermiso()) return ResultadoUbicacion.SinPermiso
        if (!gpsActivo()) return ResultadoUbicacion.GpsApagado

        return try {
            val actual = withTimeoutOrNull(12_000.milliseconds) {
                client.getCurrentLocation(
                    Priority.PRIORITY_HIGH_ACCURACY,
                    CancellationTokenSource().token
                ).await()
            }
            // Si no llegó una lectura nueva en 12 s, usa la última conocida
            val loc = actual ?: client.lastLocation.await()

            if (loc == null) {
                ResultadoUbicacion.Error("No se pudo obtener la ubicación. Sal al aire libre e intenta de nuevo.")
            } else {
                val simulada = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
                    loc.isMock
                } else {
                    @Suppress("DEPRECATION")
                    loc.isFromMockProvider
                }
                ResultadoUbicacion.Ok(
                    LecturaUbicacion(
                        latitud = loc.latitude,
                        longitud = loc.longitude,
                        precisionMetros = loc.accuracy,
                        simulada = simulada
                    )
                )
            }
        } catch (e: Exception) {
            ResultadoUbicacion.Error("Error al leer la ubicación: ${e.message}")
        }
    }
}

package com.js.smartvolt.asiste.model.data.local

import android.content.Context
import androidx.room.Room

object DatabaseProvider {
    @Volatile
    private var db: AppDatabase? = null

    private fun database(context: Context): AppDatabase {
        if (db == null) {
            db = Room.databaseBuilder(
                context.applicationContext,
                AppDatabase::class.java,
                "asiste.db"
            )
                .fallbackToDestructiveMigration(true)
                .fallbackToDestructiveMigrationOnDowngrade(true)
                .build()
        }
        return db!!
    }

    fun getAsistenciaDao(context: Context): AsistenciaDAO = database(context).asistenciaDao()

    fun getUsuarioDao(context: Context): UsuarioDAO =
        database(context).usuarioDao()

    fun getFaenaDao(context: Context): FaenaDao =
        database(context).faenaDao()

    fun getHoraExtraDao(context: Context): HoraExtraDao =
        database(context).horaExtraDao()

    fun getTurnoDao(context: Context): TurnoDAO =
        database(context).turnoDao()
    fun getAlertaDao(context: Context): AlertaDao =
        database(context).alertaDao()
    // Compatibilidad: por defecto devuelve el DAO de asistencia
    fun getDao(context: Context): AsistenciaDAO = getAsistenciaDao(context)
}

/** Dropdown genérico: sirve para faenas, roles, estados, operarios, etc. */
@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun <T> CampoDropdown(
    etiqueta: String,
    valorActual: String,
    opciones: List<T>,
    textoOpcion: (T) -> String,
    onSeleccion: (T) -> Unit
) {
    var expandido by remember { mutableStateOf(false) }

    ExposedDropdownMenuBox(expanded = expandido, onExpandedChange = { expandido = !expandido }) {
        OutlinedTextField(
            value = valorActual,
            onValueChange = {},
            readOnly = true,
            label = { Text(etiqueta) },
            trailingIcon = { ExposedDropdownMenuDefaults.TrailingIcon(expandido) },
            shape = MaterialTheme.shapes.medium,
            colors = colorsCampo(),
            modifier = Modifier.menuAnchor().fillMaxWidth()
        )
        ExposedDropdownMenu(expanded = expandido, onDismissRequest = { expandido = false }) {
            opciones.forEach { opcion ->
                DropdownMenuItem(
                    text = { Text(textoOpcion(opcion)) },
                    onClick = {
                        onSeleccion(opcion)
                        expandido = false
                    }
                )
            }
        }
    }
}

/** Diálogo de fecha. Devuelve la fecha elegida como "yyyy-MM-dd". */
@Composable
fun DialogoFecha(onDismiss: () -> Unit, onFechaElegida: (String) -> Unit) {
    val state = rememberDatePickerState()

    DatePickerDialog(
        onDismissRequest = onDismiss,
        confirmButton = {
            TextButton(onClick = {
                state.selectedDateMillis?.let {
                    onFechaElegida(
                        Instant.ofEpochMilli(it)
                            .atZone(ZoneId.of("UTC"))
                            .toLocalDate()
                            .toString()
                    )
                }
                onDismiss()
            }) { Text("OK") }
        },
        dismissButton = { TextButton(onClick = onDismiss) { Text("Cancelar") } }
    ) { DatePicker(state = state) }
}

package com.js.smartvolt.asiste.ui.components

import android.content.Context
import android.graphics.Color
import androidx.compose.runtime.Composable
import androidx.compose.runtime.DisposableEffect
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.LifecycleEventObserver
import androidx.lifecycle.compose.LocalLifecycleOwner
import org.osmdroid.tileprovider.tilesource.TileSourceFactory
import org.osmdroid.util.GeoPoint
import org.osmdroid.views.MapView
import org.osmdroid.views.overlay.CopyrightOverlay
import org.osmdroid.views.overlay.Marker
import org.osmdroid.views.overlay.Polygon

@Composable
fun OsmMapView(
    faenaLat: Double,
    faenaLng: Double,
    radioMetros: Double,
    usuarioLat: Double?,
    usuarioLng: Double?,
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    val lifecycleOwner = LocalLifecycleOwner.current

    val mapView = remember {
        MapView(context).apply {
            setTileSource(TileSourceFactory.MAPNIK)
            setMultiTouchControls(true)
            controller.setZoom(18.0)
            controller.setCenter(GeoPoint(faenaLat, faenaLng))
        }
    }

    // Overlays creados una sola vez (evita parpadeo)
    val overlayRadio = remember(mapView) { Polygon() }
    val overlayFaena = remember(mapView) {
        Marker(mapView).apply {
            setAnchor(Marker.ANCHOR_CENTER, Marker.ANCHOR_BOTTOM)
            title = "Faena"
        }
    }
    val overlayUsuario = remember(mapView) {
        Marker(mapView).apply {
            setAnchor(Marker.ANCHOR_CENTER, Marker.ANCHOR_BOTTOM)
            title = "Tu ubicación"
        }
    }

    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            when (event) {
                Lifecycle.Event.ON_RESUME -> mapView.onResume()
                Lifecycle.Event.ON_PAUSE -> mapView.onPause()
                else -> Unit
            }
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose {
            lifecycleOwner.lifecycle.removeObserver(observer)
            mapView.onDetach()
        }
    }

    // Solo recentra cuando cambia la posición del usuario
    LaunchedEffect(usuarioLat, usuarioLng) {
        if (usuarioLat != null && usuarioLng != null) {
            mapView.controller.animateTo(GeoPoint(usuarioLat, usuarioLng))
        }
    }

    AndroidView(
        modifier = modifier,
        factory = {
            mapView.apply {
                overlays.add(CopyrightOverlay(context))   // atribución obligatoria de OSM
                overlays.add(overlayRadio)
                overlays.add(overlayFaena)
                overlays.add(overlayUsuario)
            }
        },
        update = { map ->
            val centro = GeoPoint(faenaLat, faenaLng)

            overlayFaena.position = centro
            overlayRadio.points = Polygon.pointsAsCircle(centro, radioMetros)
            overlayRadio.fillPaint.color = Color.argb(45, 0, 160, 80)
            overlayRadio.outlinePaint.color = Color.argb(200, 0, 160, 80)
            overlayRadio.outlinePaint.strokeWidth = 3f

            if (usuarioLat != null && usuarioLng != null) {
                overlayUsuario.position = GeoPoint(usuarioLat, usuarioLng)
                overlayUsuario.isEnabled = true
            } else {
                overlayUsuario.isEnabled = false
            }

            map.invalidate()
        }
    )
}

@Composable
fun AsisteSVTheme(
    // true = oscuro, false = claro. Por defecto sigue el modo del sistema.
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit
) {
    // Barra de estado transparente; los íconos se adaptan al tema
    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = Color.Transparent.toArgb()
            WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = !darkTheme
        }
    }

    CompositionLocalProvider(
        LocalEstadoColors provides if (darkTheme) EstadoColorsFaena else EstadoColorsClaro,
        LocalDegradadoFondo provides if (darkTheme) DegradadoFondo else DegradadoFondoClaro
    ) {
        MaterialTheme(
            colorScheme = if (darkTheme) ColoresOscuros else ColoresClaros,
            typography = Typography,
            shapes = AppShapes,
            content = content
        )
    }
}

@Composable
fun CalendarioView(codigoOperario: String = "OP-014") {
    val context = LocalContext.current

    val dao = remember { DatabaseProvider.getDao(context) }
    val ubicacion = remember { UbicacionProvider(context) }
    val vm: AsistenciaViewModel = viewModel(
        factory = AsistenciaViewModelFactory(dao, ubicacion)
    )
    val attendances by vm.attendances.collectAsState()
    val selected by vm.selected.collectAsState()

    val formatoHora = remember { DateTimeFormatter.ofPattern("HH:mm") }
    val dias = remember(attendances) {
        attendances.mapValues { (_, asistencia) -> asistencia.aUi(formatoHora) }
    }
    val seleccionado = remember(selected) { selected?.aUi(formatoHora) }

    val usuarioDao = remember { DatabaseProvider.getUsuarioDao(context) }
    val turnoDao = remember { DatabaseProvider.getTurnoDao(context) }
    val horaExtraDao = remember { DatabaseProvider.getHoraExtraDao(context) }
    var usuarioId by remember { mutableStateOf(0L) }
    LaunchedEffect(codigoOperario) {
        val u = usuarioDao.getByCodigo(codigoOperario)
        if (u != null) usuarioId = u.id
    }
    val turnos by (if (usuarioId != 0L) turnoDao.getByUsuario(usuarioId) else turnoDao.getAll()).collectAsState(initial = emptyList())
    val horasExtra by (if (usuarioId != 0L) horaExtraDao.getByUsuario(usuarioId) else kotlinx.coroutines.flow.flowOf(emptyList())).collectAsState(initial = emptyList())
    val horasExtraFechas = remember(horasExtra) { horasExtra.map { it.fecha }.toSet() }

    var selectedDate by remember { mutableStateOf<LocalDate?>(null) }

    val cs = MaterialTheme.colorScheme

    Column(Modifier.verticalScroll(rememberScrollState())) {
        CalendarioContenido(
            dias = dias,
            seleccionado = seleccionado,
            onMesVisible = { vm.setMonth(it) },
            onSeleccionar = { selectedDate = it; vm.selectDate(it) },
            onLimpiar = { selectedDate = null; vm.clearSelection() },
            turnos = turnos.map { it.dia }.toSet(),
            horasExtra = horasExtraFechas
        )
        if (turnos.isNotEmpty()) {
            val forma = RoundedCornerShape(14.dp)
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .clip(forma)
                    .background(cs.surface)
                    .border(1.dp, cs.outline, forma)
                    .padding(12.dp),
                verticalArrangement = Arrangement.spacedBy(6.dp)
            ) {
                Text("Turnos asignados", color = cs.primary, fontSize = 14.sp, fontWeight = FontWeight.ExtraBold)
                turnos.forEach { t ->
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.SpaceBetween,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Text("${t.dia}  ${t.horario}", color = cs.onSurface, fontSize = 13.sp, fontWeight = FontWeight.SemiBold)
                        Text(if (t.activo) "Activo" else "Inactivo", color = if (t.activo) cs.primary else cs.onSurfaceVariant, fontSize = 12.sp, fontWeight = FontWeight.Medium)
                    }
                }
            }
        }
        selected?.let { sel ->
            turnos.firstOrNull { it.dia == sel.fecha }?.let { t ->
                Text("Turno del día: ${t.horario} (${if (t.activo) "Activo" else "Inactivo"})", modifier = Modifier.padding(8.dp))
            }
        }
        selectedDate?.let { d ->
            val t = turnos.firstOrNull { it.dia == d }
            if (t != null) {
                Text("Turno seleccionado: ${t.horario} (${if (t.activo) "Activo" else "Inactivo"})", modifier = Modifier.padding(8.dp))
            } else {
                Text("Día seleccionado: $d (sin turno asignado)", modifier = Modifier.padding(8.dp))
            }
        }
    }
}

// Contenido
@Composable
fun CalendarioContenido(
    dias: Map<LocalDate, DiaAsistenciaUi>,
    seleccionado: DiaAsistenciaUi?,
    onMesVisible: (YearMonth) -> Unit,
    onSeleccionar: (LocalDate) -> Unit,
    onLimpiar: () -> Unit,
    turnos: Set<LocalDate> = emptySet(),
    horasExtra: Set<LocalDate> = emptySet(),
    modifier: Modifier = Modifier,
    hoy: LocalDate = LocalDate.now()
) {
    val cs = MaterialTheme.colorScheme
    val estados = LocalEstadoColors.current
    val ctx = LocalContext.current
    val scope = rememberCoroutineScope()

    val mesInicial = remember(hoy) { YearMonth.from(hoy) }
    val diasSemana = remember { daysOfWeek(firstDayOfWeek = DayOfWeek.MONDAY) }

    val state = rememberCalendarState(
        startMonth = mesInicial.minusMonths(6),
        endMonth = mesInicial.plusMonths(6),
        firstVisibleMonth = mesInicial,
        firstDayOfWeek = DayOfWeek.MONDAY
    )
    val mesVisible = state.firstVisibleMonth.yearMonth

    // Avisa al ViewModel cuando cambia el mes visible
    LaunchedEffect(state) {
        snapshotFlow { state.firstVisibleMonth.yearMonth }.collect { visible ->
            onMesVisible(visible)
            onLimpiar()
        }
    }

    val mesTexto = mesVisible.month.getDisplayName(TextStyle.FULL, LOCALE_ES)
        .replaceFirstChar { it.uppercase() } + " " + mesVisible.year

    val delMes = dias.values.filter { YearMonth.from(it.fecha) == mesVisible }
    val nCumplidas = delMes.count { it.estado == EstadoAsistencia.CUMPLIDA }
    val nAtrasadas = delMes.count { it.estado == EstadoAsistencia.ATRASADA }
    val nFaltas = delMes.count { it.estado == EstadoAsistencia.FALTA }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(MaterialTheme.degradadoFondo)
            .navigationBarsPadding()
    ) {
        // Encabezado (respeta la barra de estado)
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .statusBarsPadding()
                .padding(horizontal = 16.dp, vertical = 14.dp)
        ) {
            Text(
                text = "MI CALENDARIO",
                color = cs.primary,
                fontSize = 18.sp,
                fontWeight = FontWeight.ExtraBold
            )
            Text(
                text = "Turnos y asistencia",
                color = cs.onSurfaceVariant,
                fontSize = 12.sp
            )
        }

        // Línea de marca
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp)
                .height(3.dp)
                .clip(RoundedCornerShape(2.dp))
                .background(DegradadoMarca)
        )

        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(14.dp)
        ) {
            // ---------- Resumen del mes ----------
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(10.dp)
            ) {
                ResumenMes("Cumplidas", nCumplidas, estados.cumplida, Modifier.weight(1f))
                ResumenMes("Atrasadas", nAtrasadas, estados.atrasada, Modifier.weight(1f))
                ResumenMes("Faltas", nFaltas, estados.falta, Modifier.weight(1f))
            }

            // ---------- Tarjeta del calendario ----------
            val formaTarjeta = RoundedCornerShape(18.dp)
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .clip(formaTarjeta)
                    .background(cs.surface)
                    .border(1.5.dp, cs.outline, formaTarjeta)
                    .padding(12.dp)
            ) {
                // Navegación de mes
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    BotonMes("‹") {
                        scope.launch { state.animateScrollToMonth(mesVisible.minusMonths(1)) }
                    }
                    Text(
                        text = mesTexto,
                        color = cs.onSurface,
                        fontSize = 20.sp,
                        fontWeight = FontWeight.ExtraBold,
                        textAlign = TextAlign.Center
                    )
                    BotonMes("›") {
                        scope.launch { state.animateScrollToMonth(mesVisible.plusMonths(1)) }
                    }
                }

                Spacer(Modifier.height(10.dp))

                // Encabezado de días de la semana (L M M J V S D)
                Row(Modifier.fillMaxWidth()) {
                    diasSemana.forEach { dow ->
                        Box(
                            modifier = Modifier.weight(1f),
                            contentAlignment = Alignment.Center
                        ) {
                            Text(
                                text = dow.getDisplayName(TextStyle.SHORT, LOCALE_ES)
                                    .take(1)
                                    .uppercase(),
                                color = cs.primary,
                                fontSize = 12.sp,
                                fontWeight = FontWeight.ExtraBold
                            )
                        }
                    }
                }

                Spacer(Modifier.height(4.dp))

                // Cuadrícula del mes
                HorizontalCalendar(
                    state = state,
                    dayContent = { day ->
                        DiaCalendario(
                            dia = day,
                            hoy = hoy,
                            registro = dias[day.date],
                            seleccionado = seleccionado?.fecha == day.date,
                            tieneTurno = turnos.contains(day.date),
                            tieneHoraExtra = horasExtra.contains(day.date),
                            onClick = {
                                val t = turnos.firstOrNull { it == day.date }

                                if (seleccionado?.fecha == day.date) onLimpiar()
                                else onSeleccionar(day.date)
                            }
                        )
                    }
                )

                Spacer(Modifier.height(8.dp))
                HorizontalDivider(color = cs.outline)
                Spacer(Modifier.height(10.dp))

                Leyenda()
            }

            // ---------- Detalle del día seleccionado ----------
            if (seleccionado != null) {
                DetalleDia(dia = seleccionado)
            } else {
                Text(
                    text = "Toca un día con turno para ver el detalle",
                    color = cs.onSurfaceVariant,
                    fontSize = 13.sp,
                    modifier = Modifier.fillMaxWidth(),
                    textAlign = TextAlign.Center
                )
                Text(text = "Los días con un punto son horas extras asignadas",
                    color = cs.onSurfaceVariant,
                    fontSize = 13.sp,
                    modifier = Modifier.fillMaxWidth(),
                    textAlign = TextAlign.Center)
            }
        }
    }
}

@Composable
private fun DiaCalendario(
    dia: CalendarDay,
    hoy: LocalDate,
    registro: DiaAsistenciaUi?,
    seleccionado: Boolean,
    tieneTurno: Boolean = false,
    tieneHoraExtra: Boolean = false,
    onClick: () -> Unit
) {
    // Días que no pertenecen al mes: celda vacía para mantener la grilla
    if (dia.position != DayPosition.MonthDate) {
        Box(Modifier.aspectRatio(1f))
        return
    }

    val cs = MaterialTheme.colorScheme
    val estados = LocalEstadoColors.current
    val ctx = LocalContext.current

    val forma = RoundedCornerShape(10.dp)
    val esHoy = dia.date == hoy
    val esFinDeSemana = dia.date.dayOfWeek == DayOfWeek.SATURDAY ||
            dia.date.dayOfWeek == DayOfWeek.SUNDAY

    // Turno próximo: día hábil futuro (o de hoy) sin marca registrada.
    val esProximo = registro == null && !esFinDeSemana && !dia.date.isBefore(hoy)

    val relleno = when {
        registro != null -> estados.de(registro.estado)
        esHoy -> ColorHoy
        esProximo -> cs.outline.copy(alpha = 0.35f)
        else -> Color.Transparent
    }
    val colorTexto = when {
        registro != null -> estados.sobreEstado
        esHoy -> TextoOscuro
        esProximo -> cs.onSurface
        else -> cs.onSurfaceVariant.copy(alpha = 0.45f)
    }

    Box(
        modifier = Modifier
            .padding(2.5.dp)
            .aspectRatio(1f)
            .clip(forma)
            .background(relleno)
            .then(
                if (tieneTurno && !dia.date.isBefore(hoy) && !esHoy) Modifier.bordeDiscontinuo(cs.onSurfaceVariant) else Modifier
            )
            .then(if (esHoy) Modifier.border(3.dp, cs.primary, forma) else Modifier)
            .then(if (seleccionado) Modifier.border(3.dp, cs.onSurface, forma) else Modifier)
            .clickable(enabled = registro != null || tieneTurno) { onClick() },
        contentAlignment = Alignment.Center
    ) {
        Column(horizontalAlignment = Alignment.CenterHorizontally) {
            Text(
                text = dia.date.dayOfMonth.toString(),
                color = colorTexto,
                fontSize = 15.sp,
                fontWeight = if (registro != null || esHoy || esProximo) FontWeight.ExtraBold
                else FontWeight.Normal
            )
            if (tieneHoraExtra) {
                Spacer(Modifier.height(2.dp))
                Box(Modifier.size(4.dp).clip(RoundedCornerShape(2.dp)).background(cs.primary))
            }
        }
    }
}
//turnos proximos
private fun Modifier.bordeDiscontinuo(
    color: Color,
    radio: Dp = 10.dp
): Modifier = this.drawBehind {
    val grosor = 1.5.dp.toPx()
    drawRoundRect(
        color = color,
        topLeft = Offset(grosor / 2, grosor / 2),
        size = Size(size.width - grosor, size.height - grosor),
        cornerRadius = CornerRadius(radio.toPx(), radio.toPx()),
        style = Stroke(
            width = grosor,
            pathEffect = PathEffect.dashPathEffect(floatArrayOf(10f, 8f), 0f)
        )
    )
}

@Composable
private fun Leyenda() {
    val estados = LocalEstadoColors.current
    Column(
        modifier = Modifier.fillMaxWidth(),
        verticalArrangement = Arrangement.spacedBy(6.dp)
    ) {
        Row(modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween) {
            ItemLeyenda("Cumplida", estados.cumplida)
            ItemLeyenda("Atrasada", estados.atrasada)
            ItemLeyenda("Falta", estados.falta)
        }
        Row(modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween) {
            ItemLeyenda("Rechazada", estados.rechazada)
            ItemLeyenda("En Revisión", estados.enRevision)
            ItemLeyenda("Próximo", null)
        }
    }
}

@Composable
private fun ItemLeyenda(texto: String, color: Color?) {
    val cs = MaterialTheme.colorScheme
    Row(verticalAlignment = Alignment.CenterVertically) {
        Box(
            modifier = Modifier
                .size(12.dp)
                .clip(RoundedCornerShape(3.dp))
                .then(
                    if (color != null) Modifier.background(color)
                    else Modifier.bordeDiscontinuo(cs.onSurfaceVariant, radio = 3.dp)
                )
        )
        Spacer(Modifier.width(5.dp))
        Text(
            text = texto,
            color = cs.onSurface,
            fontSize = 11.sp,
            fontWeight = FontWeight.Bold
        )
    }
}

@Composable
private fun DetalleDia(dia: DiaAsistenciaUi) {
    val cs = MaterialTheme.colorScheme
    val estados = LocalEstadoColors.current
    val colorEstado = estados.de(dia.estado)

    val fechaTexto = dia.fecha
        .format(DateTimeFormatter.ofPattern("EEEE d 'de' MMMM", LOCALE_ES))
        .replaceFirstChar { it.uppercase() }

    val estadoTexto = when (dia.estado) {
        EstadoAsistencia.CUMPLIDA -> "CUMPLIDA"
        EstadoAsistencia.ATRASADA -> "ATRASADA"
        EstadoAsistencia.FALTA -> "FALTA"
        EstadoAsistencia.EN_REVISION -> "EN REVISION"
        EstadoAsistencia.RECHAZADA -> "RECHAZADA"
    }

    val forma = RoundedCornerShape(16.dp)

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .height(IntrinsicSize.Min)
            .clip(forma)
            .background(cs.surface)
            .border(1.5.dp, cs.outline, forma)
    ) {
        // Barra lateral con el color del estado
        Box(
            modifier = Modifier
                .width(8.dp)
                .fillMaxHeight()
                .background(colorEstado)
        )

        Column(
            modifier = Modifier
                .weight(1f)
                .padding(horizontal = 14.dp, vertical = 12.dp)
        ) {
            Text(
                text = fechaTexto,
                color = cs.onSurface,
                fontSize = 16.sp,
                fontWeight = FontWeight.ExtraBold
            )
            Spacer(Modifier.height(4.dp))

            FilaDetalle("Turno") {
                Text(
                    text = dia.turno,
                    color = cs.onSurface,
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Bold
                )
            }
            HorizontalDivider(color = cs.outline)
            FilaDetalle("Entrada") {
                Text(
                    text = dia.entrada,
                    color = cs.onSurface,
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Bold
                )
            }
            HorizontalDivider(color = cs.outline)
            FilaDetalle("Estado") {
                Box(
                    modifier = Modifier
                        .clip(RoundedCornerShape(6.dp))
                        .background(colorEstado)
                        .padding(horizontal = 8.dp, vertical = 3.dp)
                ) {
                    Text(
                        text = estadoTexto,
                        color = estados.sobreEstado,
                        fontSize = 11.sp,
                        fontWeight = FontWeight.ExtraBold
                    )
                }
            }
        }
    }
}

private val PERMISOS_UBICACION = arrayOf(
    Manifest.permission.ACCESS_FINE_LOCATION,
    Manifest.permission.ACCESS_COARSE_LOCATION
)

private fun tienePermisoPreciso(context: Context): Boolean =
    ContextCompat.checkSelfPermission(
        context,
        Manifest.permission.ACCESS_FINE_LOCATION
    ) == PackageManager.PERMISSION_GRANTED

private fun Context.findActivity(): Activity? {
    var ctx: Context? = this
    while (ctx is ContextWrapper) {
        if (ctx is Activity) return ctx
        ctx = ctx.baseContext
    }
    return null
}

@Composable
fun InicioRoute(
    viewModel: AsistenciaViewModel,
    faena: Faena,
    codigoOperario: String,
    fecha: LocalDate,
    inicioTurno: LocalTime,
    finTurno: LocalTime,
    frente: String,
    sinConexion: Boolean,
    onResultado: (ResultadoMarca) -> Unit,
    onVerCalendario: () -> Unit = {},
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    val lifecycleOwner = LocalLifecycleOwner.current

    val presencia by viewModel.presencia.collectAsState()
    val asistenciaHoy by viewModel.asistenciaHoy.collectAsState()

    var permisoConcedido by remember { mutableStateOf(tienePermisoPreciso(context)) }
    var yaPregunto by remember { mutableStateOf(false) }

    val launcher = rememberLauncherForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { respuesta ->
        permisoConcedido = respuesta[Manifest.permission.ACCESS_FINE_LOCATION] == true
        if (permisoConcedido) viewModel.actualizarPresencia(faena)
    }

    val pedirPermiso: () -> Unit = {
        val activity = context.findActivity()
        val rationale = activity != null && ActivityCompat.shouldShowRequestPermissionRationale(
            activity,
            Manifest.permission.ACCESS_FINE_LOCATION
        )
        if (yaPregunto && !rationale) {
            // Denegado para siempre: mandar a Ajustes de la app
            context.startActivity(
                Intent(
                    Settings.ACTION_APPLICATION_DETAILS_SETTINGS,
                    Uri.fromParts("package", context.packageName, null)
                )
            )
        } else {
            yaPregunto = true
            val activity = context.findActivity()
            if (activity != null) {
                ActivityCompat.requestPermissions(activity, PERMISOS_UBICACION, 1001)
            }
        }
    }

    // Al entrar: lee el GPS o pide el permiso
    LaunchedEffect(faena.id) {
        if (tienePermisoPreciso(context)) {
            permisoConcedido = true
            viewModel.actualizarPresencia(faena)
        } else {
            permisoConcedido = false
        }
    }

    // Al volver de Ajustes, re-verifica
    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            if (event == Lifecycle.Event.ON_RESUME) {
                val ok = tienePermisoPreciso(context)
                if (ok && !permisoConcedido) {
                    permisoConcedido = true
                    viewModel.actualizarPresencia(faena)
                }
            }
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
    }

    // Entrega cada resultado una sola vez
    LaunchedEffect(Unit) {
        viewModel.resultado.collect { r ->
            if (r != null) {
                onResultado(r)
                viewModel.limpiarResultado()
            }
        }
    }

    val entradaHecha = asistenciaHoy?.estado == EstadoAsistencia.CUMPLIDA ||
            asistenciaHoy?.estado == EstadoAsistencia.ATRASADA
    val salidaHecha = asistenciaHoy?.registroSalida != null

    val textoBoton = when {
        salidaHecha -> "JORNADA CERRADA"
        entradaHecha -> "MARCAR SALIDA"
        else -> "MARCAR ENTRADA"
    }

    InicioView(
        codigoOperario = codigoOperario,
        nombreFaena = faena.nombre,
        fecha = fecha,
        inicioTurno = inicioTurno,
        finTurno = finTurno,
        frente = frente,
        validacion = ValidacionPresencia(
            gpsDentroFaena = presencia.dentroFaena,
            balizaDetectada = false,
            rssiDbm = null
        ),
        sinConexion = sinConexion,
        textoBoton = textoBoton,
        aviso = presencia.aviso,
        cargando = presencia.cargando || salidaHecha,
        onMarcarEntrada = {
            when {
                !permisoConcedido -> pedirPermiso()
                salidaHecha -> Unit
                entradaHecha -> viewModel.marcarSalida(faena, fecha, sinConexion)
                else -> viewModel.marcarEntrada(faena, fecha, inicioTurno, finTurno, sinConexion, codigoOperario)
            }
        },
        onVerCalendario = onVerCalendario,
        mapa = {
            OsmMapView(
                faenaLat = faena.latitud,
                faenaLng = faena.longitud,
                radioMetros = faena.radioMetros,
                usuarioLat = presencia.latitud,
                usuarioLng = presencia.longitud
            )
        },
        modifier = modifier
    )
}

@Composable
fun MainView(codigoOperario: String = "OP-014", onCerrarSesion: () -> Unit = {}) {
    val context = LocalContext.current
    val dao = remember { DatabaseProvider.getAsistenciaDao(context) }
    val ubicacion = remember { UbicacionProvider(context) }

    val alertaDao = DatabaseProvider.getAlertaDao(context)
    val vm: AsistenciaViewModel = viewModel(
        factory = AsistenciaViewModelFactory(dao, ubicacion, alertaDao)
    )

    val faena = remember {
        Faena(
            id = 1,
            nombre = "Faena Ruta 5 Norte",
            presentes = 42,
            total = 48,
            latitud = -33.4489,
            longitud = -70.6693,
            radioMetros = 100.0
        )
    }
    val asistenciaDao = DatabaseProvider.getAsistenciaDao(context)
    val turnoDao = DatabaseProvider.getTurnoDao(context)
    val usuarioDao = DatabaseProvider.getUsuarioDao(context)
    var selectedTab by remember { mutableStateOf(0) }
    var resultadoMarca by remember { mutableStateOf<ResultadoMarca?>(null) }
    var inicioTurno by remember { mutableStateOf(java.time.LocalTime.of(8,0)) }
    var finTurno by remember { mutableStateOf(java.time.LocalTime.of(14,0)) }
    var vistaExtra by remember { mutableStateOf<String?>(null) }
    LaunchedEffect(codigoOperario) {
        val u = usuarioDao.getByCodigo(codigoOperario)
        if (u != null) {
            val turnos = turnoDao.getByUsuario(u.id).first()
            val hoy = java.time.LocalDate.now()
            val t = turnos.firstOrNull { it.dia == hoy && it.activo }
            if (t != null) {
                inicioTurno = t.horario
                finTurno = t.horario.plusHours(6)
            }
        }
    }

    BackHandler(enabled = true) {
        if (vistaExtra != null) vistaExtra = null
        else if (selectedTab != 0) selectedTab = 0
    }

    if (resultadoMarca != null) {
        MarcaResultadoScreen(
            resultado = resultadoMarca!!,
            onAccion = { resultadoMarca = null }
        )
        return
    }

    if (vistaExtra != null) {
        HamburgerScaffold(
            title = if (vistaExtra == "config") "Configuración" else "Perfil",
            onConfiguracion = { vistaExtra = "config" },
            onPerfil = { vistaExtra = "perfil" },
            onCerrarSesion = onCerrarSesion,
            content = { padding ->
                Box(Modifier.padding(padding)) {
                    if (vistaExtra == "config") ConfiguracionView() else PerfilView(codigoOperario = codigoOperario)
                }
            }
        )
        return
    }

    HamburgerScaffold(
        title = "AsisteSV",
        onConfiguracion = { vistaExtra = "config" },
        onPerfil = { vistaExtra = "perfil" },
        onCerrarSesion = onCerrarSesion,
        bottomBar = {
            UserBottomBar(selectedTab = selectedTab, onTabSelected = { selectedTab = it })
        },
        content = { padding ->
            when (selectedTab) {
                0 -> InicioRoute(
                    viewModel = vm,
                    faena = faena,
                    codigoOperario = "OP-014",
                    fecha = LocalDate.now(),
                    inicioTurno = inicioTurno,
                    finTurno = finTurno,
                    frente = "Frente 2",
                    sinConexion = false,
                    onResultado = { resultadoMarca = it },
                    onVerCalendario = { selectedTab = 1 },
                    modifier = Modifier.padding(padding)
                )
                1 -> Box(Modifier.padding(padding)) { CalendarioView(codigoOperario = codigoOperario) }
                2 -> Box(Modifier.padding(padding)) { HistorialView() }
            }
        }
    )
}

/** Una fila de la validación de presencia (GPS, baliza BLE, ubicación simulada). */
data class CheckPresencia(
    val etiqueta: String,
    val valor: String,
    val ok: Boolean
)


/** Datos que necesita la pantalla para mostrar una marca aprobada o rechazada. */
data class ResultadoMarca(
    val aprobada: Boolean,
    val titulo: String,
    val subtitulo: String,
    val checks: List<CheckPresencia>,
    val mensaje: String,
    val nota: String? = null,
    val textoBoton: String
) {
    companion object {
        /** Marca válida. [rssi] es opcional: si no hay baliza BLE, no se muestra esa fila. */
        fun aprobada(
            hora: String,
            rssi: Int? = null,
            sinConexion: Boolean = false,
            atrasada: Boolean = false
        ): ResultadoMarca {
            val checks = buildList {
                add(CheckPresencia("GPS dentro de la faena", "SÍ", true))
                if (rssi != null) add(CheckPresencia("Baliza BLE de la faena", "$rssi dBm", true))
                add(CheckPresencia("Ubicación simulada", "NO", true))
            }
            return ResultadoMarca(
                aprobada = true,
                titulo = "MARCA REGISTRADA",
                subtitulo = "Entrada registrada a las $hora",
                checks = checks,
                mensaje = if (atrasada) "Tu asistencia quedó guardada con atraso."
                else "Tu asistencia quedó guardada correctamente.",
                nota = when {
                    sinConexion -> "Sin conexión · se guardará en el teléfono"
                    atrasada -> "Marca fuera de la tolerancia del turno"
                    else -> null
                },
                textoBoton = "VOLVER AL INICIO"
            )
        }

        /** Marca rechazada por ubicación simulada (Fake GPS). */
        fun rechazadaPorFakeGps(gpsEnFaena: Boolean = true) = ResultadoMarca(
            aprobada = false,
            titulo = "MARCA RECHAZADA",
            subtitulo = "Se detectó una ubicación simulada",
            checks = listOf(
                CheckPresencia("GPS en faena", if (gpsEnFaena) "SÍ" else "NO", gpsEnFaena),
                CheckPresencia("Baliza BLE", "NO DETECTADA", false),
                CheckPresencia("Mock Location", "ACTIVO", false)
            ),
            mensaje = "Desactiva las ubicaciones simuladas y el modo desarrollador. " +
                    "Luego acércate a la baliza de tu frente.",
            nota = "El intento quedó registrado.",
            textoBoton = "REINTENTAR"
        )

        /** Marca rechazada porque el operario está fuera del radio de la faena. */
        fun rechazadaFueraDeFaena(distanciaMetros: Int) = ResultadoMarca(
            aprobada = false,
            titulo = "MARCA RECHAZADA",
            subtitulo = "Estás fuera de la faena",
            checks = listOf(
                CheckPresencia("GPS dentro de la faena", "NO · A $distanciaMetros m", false),
                CheckPresencia("Ubicación simulada", "NO", true)
            ),
            mensaje = "Acércate a la faena y vuelve a intentar.",
            nota = "El intento quedó registrado.",
            textoBoton = "REINTENTAR"
        )

        /** No se pudo leer el GPS (sin permiso, GPS apagado o error). No se guarda nada. */
        fun sinUbicacion(motivo: String) = ResultadoMarca(
            aprobada = false,
            titulo = "SIN UBICACIÓN",
            subtitulo = "No se pudo leer tu ubicación",
            checks = listOf(
                CheckPresencia("GPS dentro de la faena", "SIN DATOS", false)
            ),
            mensaje = motivo,
            nota = null,
            textoBoton = "REINTENTAR"
        )
        /** Intenta marcar salida sin haber marcado entrada. */
        fun sinEntrada() = ResultadoMarca(
            aprobada = false,
            titulo = "SIN ENTRADA",
            subtitulo = "No registraste tu entrada hoy",
            checks = listOf(
                CheckPresencia("Entrada del día", "NO REGISTRADA", false)
            ),
            mensaje = "Primero marca tu entrada y luego la salida.",
            nota = null,
            textoBoton = "ENTENDIDO"
        )

        /** Salida registrada correctamente. */
        fun salidaRegistrada(hora: String) = ResultadoMarca(
            aprobada = true,
            titulo = "SALIDA REGISTRADA",
            subtitulo = "Salida registrada a las $hora",
            checks = listOf(
                CheckPresencia("GPS dentro de la faena", "SÍ", true),
                CheckPresencia("Ubicación simulada", "NO", true)
            ),
            mensaje = "Tu jornada quedó cerrada correctamente.",
            nota = null,
            textoBoton = "VOLVER AL INICIO"
        )
    }
}

@Composable
private fun IconoResultadoAnimado(
    aprobada: Boolean,
    color: Color,
    colorSimbolo: Color,
    clave: Any? = null,
    modifier: Modifier = Modifier
) {
    // Círculo principal
    val escala = remember { Animatable(0f) }
    val alfaCirculo = remember { Animatable(0f) }

    // Halo que se expande y desaparece
    val escalaHalo = remember { Animatable(0.6f) }
    val alfaHalo = remember { Animatable(0f) }

    // Progreso del trazo del símbolo (0 = nada, 1 = completo)
    val progreso = remember { Animatable(0f) }

    LaunchedEffect(aprobada, clave) {
        // Reinicio por si el resultado cambia
        escala.snapTo(0f)
        alfaCirculo.snapTo(0f)
        escalaHalo.snapTo(0.6f)
        alfaHalo.snapTo(0f)
        progreso.snapTo(0f)

        // 1) El círculo aparece con rebote
        launch {
            alfaCirculo.animateTo(1f, tween(durationMillis = 160))
            escala.animateTo(
                targetValue = 1f,
                animationSpec = spring(
                    dampingRatio = 0.45f,
                    stiffness = Spring.StiffnessMediumLow
                )
            )
        }

        // 2) El halo se expande y se desvanece
        launch {
            alfaHalo.animateTo(0.45f, tween(durationMillis = 140))
            escalaHalo.animateTo(
                targetValue = 1.18f,
                animationSpec = tween(durationMillis = 600, easing = FastOutSlowInEasing)
            )
            alfaHalo.animateTo(0f, tween(durationMillis = 320))
        }

        // 3) El trazo arranca cuando el círculo ya se ve
        delay(180)
        progreso.animateTo(
            targetValue = 1f,
            animationSpec = tween(durationMillis = 480, easing = FastOutSlowInEasing)
        )
    }

    Box(modifier = modifier, contentAlignment = Alignment.Center) {

        // Halo
        Box(
            Modifier
                .fillMaxSize()
                .graphicsLayer {
                    scaleX = escalaHalo.value
                    scaleY = escalaHalo.value
                    alpha = alfaHalo.value
                }
                .clip(CircleShape)
                .background(color)
        )

        // Círculo sólido (92 de 120 = 0.7667)
        Box(
            Modifier
                .fillMaxSize(0.7667f)
                .graphicsLayer {
                    scaleX = escala.value
                    scaleY = escala.value
                    alpha = alfaCirculo.value
                }
                .clip(CircleShape)
                .background(color)
        )

        // Símbolo dibujado
        Canvas(
            Modifier
                .fillMaxSize(0.5f)
                .graphicsLayer { alpha = alfaCirculo.value }
        ) {
            dibujarSimbolo(
                aprobada = aprobada,
                progreso = progreso.value,
                color = colorSimbolo,
                grosor = size.width * 0.13f
            )
        }
    }
}

/** Dibuja un ✓ o un ✕ de forma progresiva. */
private fun DrawScope.dibujarSimbolo(
    aprobada: Boolean,
    progreso: Float,
    color: Color,
    grosor: Float
) {
    val w = size.width
    val h = size.height

    // Dibuja el tramo [inicio, fin] del progreso total
    fun tramo(desde: Offset, hasta: Offset, inicio: Float, fin: Float) {
        if (progreso <= inicio) return
        val t = ((progreso - inicio) / (fin - inicio)).coerceIn(0f, 1f)
        val finParcial = Offset(
            desde.x + (hasta.x - desde.x) * t,
            desde.y + (hasta.y - desde.y) * t
        )
        drawLine(
            color = color,
            start = desde,
            end = finParcial,
            strokeWidth = grosor,
            cap = StrokeCap.Round
        )
    }

    if (aprobada) {
        // ✓ : primero la bajada corta, luego la subida larga
        val p1 = Offset(w * 0.12f, h * 0.54f)
        val p2 = Offset(w * 0.40f, h * 0.82f)
        val p3 = Offset(w * 0.90f, h * 0.18f)
        tramo(p1, p2, 0f, 0.45f)
        tramo(p2, p3, 0.45f, 1f)
    } else {
        // ✕ : primero la diagonal \
        val a = Offset(w * 0.16f, h * 0.16f)
        val b = Offset(w * 0.84f, h * 0.84f)
        val c = Offset(w * 0.84f, h * 0.16f)
        val d = Offset(w * 0.16f, h * 0.84f)
        tramo(a, b, 0f, 0.5f)
        tramo(c, d, 0.5f, 1f)
    }
}

@Composable
fun AdminHorasExtraView(codigo: String, onBack: () -> Unit) {
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    val horaExtraDao = DatabaseProvider.getHoraExtraDao(context)
    val operario = rememberSeleccionOperario(codigo)

    var fecha by remember { mutableStateOf(LocalDate.now().toString()) }
    var horas by remember { mutableStateOf("1.0") }
    var motivo by remember { mutableStateOf("") }
    var nota by remember { mutableStateOf("") }
    var estado by remember { mutableStateOf(EstadoHorasExtras.PENDIENTE) }
    var mostrarFecha by remember { mutableStateOf(false) }

    val horasExtra by (if (operario.id != 0L) horaExtraDao.getByUsuario(operario.id) else flowOf(emptyList()))
        .collectAsState(initial = emptyList())

    PantallaAdmin("HORAS EXTRA", operario.codigo, onBack) {
        Panel {
            CampoDropdown(
                etiqueta = "Operario",
                valorActual = operario.codigo,
                opciones = operario.operarios,
                textoOpcion = { it.codigo },
                onSeleccion = { operario.elegir(it) }
            )

            Spacer(Modifier.height(Medidas.EspacioM))

            Row(horizontalArrangement = Arrangement.spacedBy(Medidas.EspacioM)) {
                Box(Modifier.weight(1f)) {
                    CampoSelector("FECHA", fecha.ifBlank { "Seleccionar fecha" }) { mostrarFecha = true }
                }
                Box(Modifier.weight(1f)) {
                    CampoTexto("HORAS", horas, { horas = it })
                }
            }

            Spacer(Modifier.height(Medidas.EspacioM))
            CampoTexto("Motivo", motivo, { motivo = it })

            Spacer(Modifier.height(Medidas.EspacioM))
            CampoDropdown(
                etiqueta = "ESTADO",
                valorActual = estado.name,
                opciones = EstadoHorasExtras.values().toList(),
                textoOpcion = { it.name },
                onSeleccion = { estado = it }
            )

            Spacer(Modifier.height(Medidas.EspacioM))
            CampoTexto("Nota", nota, { nota = it }, singleLine = false)
        }

        Spacer(Modifier.height(Medidas.EspacioL))

        BotonPrincipal(
            texto = "GUARDAR HORA EXTRA",
            enabled = operario.id != 0L && horas.toDoubleOrNull() != null
        ) {
            scope.launch {
                horaExtraDao.upsert(
                    HoraExtra(
                        usuarioId = operario.id,
                        fecha = LocalDate.parse(fecha),
                        horas = horas.toDoubleOrNull() ?: 0.0,
                        motivo = motivo,
                        estado = estado,
                        nota = nota
                    )
                )
            }
            motivo = ""
            nota = ""
        }

        Spacer(Modifier.height(Medidas.EspacioL))

        TituloSeccion("HORAS EXTRA REGISTRADAS")

        if (horasExtra.isEmpty()) {
            TextoVacio("Este operario aún no tiene horas extra")
        } else {
            horasExtra.sortedByDescending { it.fecha }.forEach { h ->
                HoraExtraItem(
                    h = h,
                    onCambiarEstado = { nuevo -> scope.launch { horaExtraDao.upsert(h.copy(estado = nuevo)) } },
                    onEliminar = { scope.launch { horaExtraDao.delete(h) } }
                )
            }
        }
    }

    if (mostrarFecha) {
        DialogoFecha(onDismiss = { mostrarFecha = false }, onFechaElegida = { fecha = it })
    }
}

@Composable
private fun HoraExtraItem(
    h: HoraExtra,
    onCambiarEstado: (EstadoHorasExtras) -> Unit,
    onEliminar: () -> Unit
) {
    val color = when (h.estado) {
        EstadoHorasExtras.APROBADO -> MaterialTheme.estados.cumplida
        EstadoHorasExtras.RECHAZADO -> MaterialTheme.estados.rechazada
        else -> MaterialTheme.estados.enRevision
    }

    Panel(Modifier.padding(bottom = 10.dp)) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            Column(Modifier.weight(1f)) {
                Text(
                    text = "${h.fecha} · ${h.horas} h",
                    style = MaterialTheme.typography.titleMedium,
                    color = MaterialTheme.colorScheme.onSurface
                )
                Text(
                    text = h.motivo.ifBlank { "Sin motivo" },
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                if (h.nota.isNotBlank()) {
                    Text(
                        text = h.nota,
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
            }
            if (h.estado == EstadoHorasExtras.PENDIENTE) {
                Row(horizontalArrangement = Arrangement.spacedBy(4.dp)) {
                    TextButton(onClick = { onCambiarEstado(EstadoHorasExtras.APROBADO) }) { Text("Aprobar") }
                    TextButton(onClick = { onCambiarEstado(EstadoHorasExtras.RECHAZADO) }) { Text("Rechazar") }
                }
            } else {
                BadgeEstado(h.estado.name, color)
            }
            IconButton(onClick = onEliminar) {
                Icon(
                    imageVector = Icons.Filled.Delete,
                    contentDescription = "Eliminar hora extra",
                    tint = MaterialTheme.estados.rechazada
                )
            }
        }
    }
}

package com.js.smartvolt.asiste.ui.view.admin

import androidx.activity.compose.BackHandler

import androidx.compose.material.icons.Icons

sealed class AdminScreen {
    object Panel : AdminScreen()
    object Reportes : AdminScreen()
    object Alertas : AdminScreen()
    object Usuarios : AdminScreen()
    data class DetalleAlerta(val id: String) : AdminScreen()
    data class PerfilUsuario(val codigo: String) : AdminScreen()
    object NuevoUsuario : AdminScreen()
    data class Turnos(val codigo: String) : AdminScreen()
    data class HorasExtra(val codigo: String) : AdminScreen()
    data class DetalleOperario(val codigo: String) : AdminScreen()
}

@Composable
fun AdminMainView(onCerrarSesion: () -> Unit = {}) {
    var current by remember { mutableStateOf<AdminScreen>(AdminScreen.Panel) }
    val tabFor: (AdminScreen) -> Int = { s ->
        when (s) {
            is AdminScreen.Panel, is AdminScreen.DetalleAlerta -> 0
            is AdminScreen.Reportes, is AdminScreen.DetalleOperario -> 1
            is AdminScreen.Alertas -> 2
            is AdminScreen.Usuarios, is AdminScreen.PerfilUsuario, is AdminScreen.NuevoUsuario,
            is AdminScreen.Turnos, is AdminScreen.HorasExtra -> 3
        }
    }
    var selectedTab by remember { mutableStateOf(tabFor(current)) }
    LaunchedEffect(current) { selectedTab = tabFor(current) }

    BackHandler(enabled = true) {
        when (current) {
            is AdminScreen.DetalleAlerta -> current = AdminScreen.Alertas
            is AdminScreen.PerfilUsuario -> current = AdminScreen.Usuarios
            is AdminScreen.NuevoUsuario -> current = AdminScreen.Usuarios
            is AdminScreen.Turnos -> current = AdminScreen.PerfilUsuario((current as AdminScreen.Turnos).codigo)
            is AdminScreen.HorasExtra -> current = AdminScreen.PerfilUsuario((current as AdminScreen.HorasExtra).codigo)
            is AdminScreen.DetalleOperario -> current = AdminScreen.Reportes
            else -> if (selectedTab != 0) { selectedTab = 0; current = AdminScreen.Panel }
        }
    }

    HamburgerScaffold(
        title = "AsisteSV Admin",
        onConfiguracion = { },
        onPerfil = { },
        onCerrarSesion = onCerrarSesion,
        bottomBar = {
            AdminBottomBar(selectedTab = selectedTab, onTabSelected = { t ->
                current = when (t) {
                    0 -> AdminScreen.Panel
                    1 -> AdminScreen.Reportes
                    2 -> AdminScreen.Alertas
                    else -> AdminScreen.Usuarios
                }
            })
        },
        content = { padding ->
            Box(Modifier.padding(padding)) {
                when (val s = current) {
                    AdminScreen.Panel -> AdminPanelView(onOpenAlertas = { current = AdminScreen.Alertas }, onOpenUsuarios = { current = AdminScreen.Usuarios }, onOpenReporte = { current = AdminScreen.Reportes })
                    AdminScreen.Reportes -> AdminReportesView(onOpenOperario = { current = AdminScreen.DetalleOperario(it) })
                    AdminScreen.Alertas -> AdminAlertasView(onOpenDetalle = { current = AdminScreen.DetalleAlerta(it) })
                    AdminScreen.Usuarios -> AdminUsuariosView(onNew = { current = AdminScreen.NuevoUsuario }, onOpenPerfil = { current = AdminScreen.PerfilUsuario(it) })
                    is AdminScreen.DetalleAlerta -> AdminDetalleAlertaView(id = s.id, onBack = { current = AdminScreen.Alertas })
                    is AdminScreen.PerfilUsuario -> AdminPerfilUsuarioView(codigo = s.codigo, onBack = { current = AdminScreen.Usuarios }, onTurnos = { current = AdminScreen.Turnos(s.codigo) }, onHorasExtra = { current = AdminScreen.HorasExtra(s.codigo) })
                    AdminScreen.NuevoUsuario -> AdminNuevoUsuarioView(onBack = { current = AdminScreen.Usuarios })
                    is AdminScreen.Turnos -> AdminTurnosView(codigo = s.codigo, onBack = { current = AdminScreen.PerfilUsuario(s.codigo) })
                    is AdminScreen.HorasExtra -> AdminHorasExtraView(codigo = s.codigo, onBack = { current = AdminScreen.PerfilUsuario(s.codigo) })
                    is AdminScreen.DetalleOperario -> AdminDetalleOperarioView(codigo = s.codigo, onBack = { current = AdminScreen.Reportes })
                }
            }
        }
    )
}

private data class ResumenOperario(
    val codigo: String,
    val horas: Int,
    val extras: Double,
    val atrasos: Int,
    val faltas: Int,
    val registros: Int
)

private const val HORAS_POR_JORNADA = 8

private fun Double.formatoHoras(): String =
    if (this % 1.0 == 0.0) toInt().toString() else "%.1f".format(this)

// ---------------------------------------------------------------
// Pantalla
// ---------------------------------------------------------------
@Composable
fun AdminReportesView(onOpenOperario: (String) -> Unit) {
    val context = LocalContext.current
    val usuarioDao = DatabaseProvider.getUsuarioDao(context)
    val asistenciaDao = DatabaseProvider.getAsistenciaDao(context)
    val horaExtraDao = DatabaseProvider.getHoraExtraDao(context)
    val faenaDao = DatabaseProvider.getFaenaDao(context)
    val faenas by faenaDao.getAll().collectAsState(initial = emptyList())
    val usuarios by usuarioDao.getByRol(Rol.OPERARIO).collectAsState(initial = emptyList())

    var faenaSeleccionada by remember { mutableStateOf<Long?>(null) } // null = todas
    var mesSeleccionado by remember { mutableStateOf(java.time.YearMonth.now()) }
    var expandFaena by remember { mutableStateOf(false) }
    var expandMes by remember { mutableStateOf(false) }

    val usuariosFiltrados = if (faenaSeleccionada == null) usuarios else usuarios.filter { it.faenaId == faenaSeleccionada }

    val resumenes = usuariosFiltrados.map { u ->
        key(u.id) {
            val asistencias by asistenciaDao.getByUsuario(u.id).collectAsState(initial = emptyList())
            val horasExtra by horaExtraDao.getByUsuario(u.id).collectAsState(initial = emptyList())

            val asistMes = asistencias.filter { java.time.YearMonth.from(it.fecha) == mesSeleccionado }
            val extrasMes = horasExtra.filter { java.time.YearMonth.from(it.fecha) == mesSeleccionado && it.estado.name == "APROBADO" }

            val cumplidas = asistMes.count { it.estado == EstadoAsistencia.CUMPLIDA }
            val atrasos = asistMes.count { it.estado == EstadoAsistencia.ATRASADA }
            val faltas = asistMes.count { it.estado == EstadoAsistencia.FALTA }
            val extras = extrasMes.sumOf { it.horas.toDouble() }

            ResumenOperario(
                codigo = u.codigo,
                horas = cumplidas * HORAS_POR_JORNADA,
                extras = extras,
                atrasos = atrasos,
                faltas = faltas,
                registros = cumplidas + atrasos + faltas
            )
        }
    }
    // Un resumen por operario (cada uno lee sus propios datos)
    val totalHoras = resumenes.sumOf { it.horas }
    val totalExtras = resumenes.sumOf { it.extras }
    val totalAtrasos = resumenes.sumOf { it.atrasos }
    val totalFaltas = resumenes.sumOf { it.faltas }
    val totalRegistros = resumenes.sumOf { it.registros }
    val ausentismo = if (totalRegistros == 0) 0f else totalFaltas.toFloat() / totalRegistros

    Column(
        Modifier
            .fillMaxSize()
            .verticalScroll(rememberScrollState())
            .padding(horizontal = Medidas.PaddingPantalla, vertical = Medidas.EspacioM)
    ) {
        AdminHeader("REPORTE DE HORAS", "Remuneraciones y ausentismo")

        Spacer(Modifier.height(Medidas.EspacioM))

        Row(horizontalArrangement = Arrangement.spacedBy(Medidas.EspacioS)) {
            Box {
                SelectorChip(if (faenaSeleccionada == null) "Todas las faenas" else faenas.firstOrNull { it.id == faenaSeleccionada }?.nombre ?: "Faena", onClick = { expandFaena = true })
                DropdownMenu(expanded = expandFaena, onDismissRequest = { expandFaena = false }) {
                    DropdownMenuItem(text = { Text("Todas las faenas") }, onClick = { faenaSeleccionada = null; expandFaena = false })
                    faenas.forEach { f ->
                        DropdownMenuItem(text = { Text(f.nombre) }, onClick = { faenaSeleccionada = f.id; expandFaena = false })
                    }
                }
                // click area
            }
            Box {
                SelectorChip(mesSeleccionado.toString(), onClick = { expandMes = true })
                DropdownMenu(expanded = expandMes, onDismissRequest = { expandMes = false }) {
                    (0..5).map { mesSeleccionado.minusMonths(it.toLong()) }.forEach { m ->
                        DropdownMenuItem(text = { Text(m.toString()) }, onClick = { mesSeleccionado = m; expandMes = false })
                    }
                }
            }
        }

        Spacer(Modifier.height(Medidas.EspacioL))

        // Resumen general
        Row(
            Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(Medidas.EspacioS)
        ) {
            StatTile("HORAS", totalHoras.toString(), MaterialTheme.estados.cumplida, Modifier.weight(1f))
            StatTile("EXTRA", totalExtras.formatoHoras(), MaterialTheme.colorScheme.primary, Modifier.weight(1f))
            StatTile("ATRASOS", totalAtrasos.toString(), MaterialTheme.estados.atrasada, Modifier.weight(1f))
            StatTile("FALTAS", totalFaltas.toString(), MaterialTheme.estados.falta, Modifier.weight(1f))
        }

        Spacer(Modifier.height(Medidas.EspacioL))

        // Tabla por operario
        Card(
            modifier = Modifier.fillMaxWidth(),
            shape = MaterialTheme.shapes.large,
            colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface),
            border = BorderStroke(Medidas.BordeCard, MaterialTheme.colorScheme.outline)
        ) {
            Column(Modifier.padding(vertical = Medidas.EspacioS)) {
                FilaEncabezado()
                HorizontalDivider(color = MaterialTheme.colorScheme.outline)

                if (resumenes.isEmpty()) {
                    Text(
                        text = "No hay operarios registrados",
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(Medidas.PaddingCard),
                        textAlign = TextAlign.Center
                    )
                } else {
                    resumenes.forEachIndexed { index, r ->
                        FilaOperario(r, onClick = { onOpenOperario(r.codigo) })
                        if (index < resumenes.lastIndex) {
                            HorizontalDivider(
                                modifier = Modifier.padding(horizontal = Medidas.PaddingCard),
                                color = MaterialTheme.colorScheme.outline.tenue(0.5f)
                            )
                        }
                    }
                }

                HorizontalDivider(color = MaterialTheme.colorScheme.outline)
                FilaTotal(totalHoras, totalExtras, totalAtrasos, totalFaltas)
            }
        }

        Spacer(Modifier.height(Medidas.EspacioL))

        // Ausentismo
        AusentismoCard(ausentismo)

        Spacer(Modifier.height(Medidas.EspacioL))

        Button(
            onClick = { },
            modifier = Modifier
                .fillMaxWidth()
                .height(Medidas.AlturaBoton),
            shape = MaterialTheme.shapes.medium
        ) {
            Text("EXPORTAR  CSV · PDF", style = MaterialTheme.typography.labelLarge)
        }
    }
}

fun AdminTurnosView(codigo: String, onBack: () -> Unit) {

    var activo by remember { mutableStateOf(true) }
    var mostrarFecha by remember { mutableStateOf(false) }

    val turnos by (if (operario.id != 0L) turnoDao.getByUsuario(operario.id) else flowOf(emptyList()))
        .collectAsState(initial = emptyList())

    PantallaAdmin("ASIGNAR TURNOS", operario.codigo, onBack) {

                Box(Modifier.weight(1f)) {
                    CampoDropdown(
                        etiqueta = "TURNO",
                        valorActual = HORARIOS.firstOrNull { it.first == horario }?.second ?: horario,

                        onSeleccion = { horario = it.first }
                    )
                }
            }

            Spacer(Modifier.height(Medidas.EspacioM))

            Row(
                Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {

                    text = "Turno activo",
                    style = MaterialTheme.typography.titleMedium,
                    color = MaterialTheme.colorScheme.onSurface

            }
        }

        Spacer(Modifier.height(Medidas.EspacioL))

        BotonPrincipal("GUARDAR TURNO", enabled = operario.id != 0L) {
            scope.launch {
                turnoDao.upsert(

                        usuarioId = operario.id,
                        dia = LocalDate.parse(dia),
                        activo = activo,
                        horario = LocalTime.parse(horario)
                    )
                )
            }
        }

    }
}

package com.js.smartvolt.asiste.ui.view.admin

import androidx.compose.runtime.*
import androidx.compose.ui.platform.LocalContext
import com.js.smartvolt.asiste.model.admin.Rol
import com.js.smartvolt.asiste.model.admin.Usuario
import com.js.smartvolt.asiste.model.data.local.DatabaseProvider

/** Operario seleccionado en pantallas de turnos y horas extra. */
class SeleccionOperario(codigoInicial: String) {
    var id by mutableStateOf(0L)
    var codigo by mutableStateOf(codigoInicial)
    var operarios by mutableStateOf<List<Usuario>>(emptyList())

    fun elegir(u: Usuario) {
        id = u.id
        codigo = u.codigo
    }
}

/** Carga los operarios y deja seleccionado el que viene por parámetro (o el primero). */
@Composable
fun rememberSeleccionOperario(codigoInicial: String): SeleccionOperario {
    val context = LocalContext.current
    val seleccion = remember { SeleccionOperario(codigoInicial) }
    val flujo = remember { DatabaseProvider.getUsuarioDao(context).getByRol(Rol.OPERARIO) }
    val operarios by flujo.collectAsState(initial = emptyList())

    LaunchedEffect(operarios) {
        seleccion.operarios = operarios
        if (seleccion.id == 0L && operarios.isNotEmpty()) {
            seleccion.elegir(operarios.firstOrNull { it.codigo == codigoInicial } ?: operarios.first())
        }
    }
    return seleccion
}

    data class PerfilUsuario(val codigo: String) : CapatazScreen()
    data class Turnos(val codigo: String) : CapatazScreen()
    data class HorasExtra(val codigo: String) : CapatazScreen()
    data class DetalleOperario(val codigo: String) : CapatazScreen()

@Composable
fun CapatazMainView(onCerrarSesion: () -> Unit = {}) {
    var current by remember { mutableStateOf<CapatazScreen>(CapatazScreen.Panel) }
    val tabFor: (CapatazScreen) -> Int = { s ->
        when (s) {
            is CapatazScreen.Panel, is CapatazScreen.DetalleAlerta -> 0
            is CapatazScreen.Reportes, is CapatazScreen.DetalleOperario -> 1
            is CapatazScreen.Alertas -> 2
            else -> 3
        }
    }
    var selectedTab by remember { mutableStateOf(tabFor(current)) }
    LaunchedEffect(current) { selectedTab = tabFor(current) }

    BackHandler(enabled = true) {
        when (current) {
            is CapatazScreen.DetalleAlerta -> current = CapatazScreen.Alertas
            is CapatazScreen.PerfilUsuario -> current = CapatazScreen.Panel
            is CapatazScreen.Turnos -> current = CapatazScreen.PerfilUsuario((current as CapatazScreen.Turnos).codigo)
            is CapatazScreen.HorasExtra -> current = CapatazScreen.PerfilUsuario((current as CapatazScreen.HorasExtra).codigo)
            is CapatazScreen.DetalleOperario -> current = CapatazScreen.Reportes
            else -> if (selectedTab != 0) { selectedTab = 0; current = CapatazScreen.Panel }
        }
    }

    HamburgerScaffold(
        title = "AsisteSV Capataz",
        onConfiguracion = { },
        onPerfil = { },
        onCerrarSesion = onCerrarSesion,
        bottomBar = {

                current = when (t) {
                    0 -> CapatazScreen.Panel
                    1 -> CapatazScreen.Reportes
                    2 -> CapatazScreen.Alertas
                    else -> CapatazScreen.Usuarios
                }
            })
        },
        content = { padding ->
            Box(Modifier.padding(padding)) {
                when (val s = current) {
                    CapatazScreen.Panel -> CapatazPanelView(onOpenAlertas = { current = CapatazScreen.Alertas }, onOpenOperarios = { current = CapatazScreen.Usuarios }, onOpenReporte = { current = CapatazScreen.Reportes })
                    CapatazScreen.Usuarios -> AdminUsuariosView(
                        onNew = { },
                        onOpenPerfil = { current = CapatazScreen.PerfilUsuario(it) },
                        allowNew = false
                    )
                    CapatazScreen.Reportes -> AdminReportesView(onOpenOperario = {
                        current = CapatazScreen.DetalleOperario(it)
                    })
                    CapatazScreen.Alertas -> AdminAlertasView(onOpenDetalle = {
                        current = CapatazScreen.DetalleAlerta(it)
                    })
                    is CapatazScreen.DetalleAlerta -> AdminDetalleAlertaView(
                        id = s.id,
                        onBack = { current = CapatazScreen.Alertas })
                    is CapatazScreen.PerfilUsuario -> AdminPerfilUsuarioView(

                        onBack = { current = CapatazScreen.Panel },

                        onHorasExtra = { current = CapatazScreen.HorasExtra(s.codigo) },

                    is CapatazScreen.DetalleOperario -> AdminDetalleOperarioView(

                }
            }
        }
    )

private const val TOLERANCIA_MINUTOS = 10L
private val FORMATO_HORA: DateTimeFormatter = DateTimeFormatter.ofPattern("HH:mm")

data class PresenciaUi(
    val cargando: Boolean = false,
    val latitud: Double? = null,
    val longitud: Double? = null,
    val precisionMetros: Float? = null,
    val distanciaMetros: Float? = null,
    val dentroFaena: Boolean = false,
    val simulada: Boolean = false,
    val aviso: String? = null
)

@OptIn(ExperimentalCoroutinesApi::class)
class AsistenciaViewModel(
    private val dao: AsistenciaDAO,
    private val ubicacion: UbicacionProvider,
    private val alertaDao: com.js.smartvolt.asiste.model.data.local.AlertaDao? = null
) : ViewModel() {

    // ---------- Calendario ----------
    private val _month = MutableStateFlow(YearMonth.now())
    val month: StateFlow<YearMonth> = _month.asStateFlow()

    private val _attendances = MutableStateFlow<Map<LocalDate, Asistencia>>(emptyMap())
    val attendances: StateFlow<Map<LocalDate, Asistencia>> = _attendances.asStateFlow()

    private val _selected = MutableStateFlow<Asistencia?>(null)
    val selected: StateFlow<Asistencia?> = _selected.asStateFlow()

    // ---------- Marca con GPS ----------
    private val _presencia = MutableStateFlow(PresenciaUi())
    val presencia: StateFlow<PresenciaUi> = _presencia.asStateFlow()

    private val _resultado = MutableStateFlow<ResultadoMarca?>(null)
    val resultado: StateFlow<ResultadoMarca?> = _resultado.asStateFlow()

    private val _asistenciaHoy = MutableStateFlow<Asistencia?>(null)
    val asistenciaHoy: StateFlow<Asistencia?> = _asistenciaHoy.asStateFlow()

    private var marcando = false

    init {
        viewModelScope.launch {
            _month.flatMapLatest { m ->
                dao.getByRange(m.atDay(1), m.atEndOfMonth()).map { list ->
                    list.associateBy { it.fecha }
                }
            }.collect { _attendances.value = it }
        }
        viewModelScope.launch {
            _asistenciaHoy.value = dao.getByFecha(LocalDate.now()).lastOrNull()
        }
    }

    fun setMonth(m: YearMonth) { _month.value = m }
    fun selectDate(date: LocalDate) { _selected.value = _attendances.value[date] }
    fun clearSelection() { _selected.value = null }
    fun limpiarResultado() { _resultado.value = null }

    fun actualizarPresencia(faena: Faena) {
        viewModelScope.launch {
            _presencia.value = _presencia.value.copy(cargando = true, aviso = null)
            _presencia.value = leerPresencia(faena)
        }
    }

    fun marcarEntrada(
        faena: Faena,
        fecha: LocalDate,
        inicioTurno: LocalTime,
        finTurno: LocalTime,
        sinConexion: Boolean = false,
        codigoOperario: String = "OP-014"
    ) {
        if (marcando) return
        marcando = true
        viewModelScope.launch {
            try {
                val yaMarcada = dao.getByFecha(fecha).firstOrNull {
                    it.estado == EstadoAsistencia.CUMPLIDA || it.estado == EstadoAsistencia.ATRASADA
                }
                if (yaMarcada != null) {
                    _resultado.value = ResultadoMarca.aprobada(
                        hora = yaMarcada.registro?.format(FORMATO_HORA) ?: "--:--",
                        sinConexion = sinConexion
                    )
                    return@launch
                }

                _presencia.value = _presencia.value.copy(cargando = true, aviso = null)
                val lectura = leerPresencia(faena)
                _presencia.value = lectura

                val lat = lectura.latitud
                val lng = lectura.longitud
                val registro = LocalTime.now().withSecond(0).withNano(0)

                when {
                    lat == null || lng == null ->
                        _resultado.value = ResultadoMarca.sinUbicacion(
                            lectura.aviso ?: "No se pudo leer tu ubicación."
                        )

                    lectura.simulada -> {
                        dao.upsert(nuevaAsistencia(fecha, EstadoAsistencia.RECHAZADA, inicioTurno, finTurno, registro, lat, lng))
                        alertaDao?.upsert(com.js.smartvolt.asiste.model.admin.Alerta(
                            codigoOperario = codigoOperario,
                            tipoMarca = com.js.smartvolt.asiste.model.admin.TipoMarca.ENTRADA,
                            hora = registro,
                            faenaId = faena.id,
                            motivo = "Ubicación simulada",
                            estado = "REVISION"
                        ))
                        _resultado.value = ResultadoMarca.rechazadaPorFakeGps(lectura.dentroFaena)
                    }

                    !lectura.dentroFaena -> {
                        dao.upsert(nuevaAsistencia(fecha, EstadoAsistencia.RECHAZADA, inicioTurno, finTurno, registro, lat, lng))
                        alertaDao?.upsert(com.js.smartvolt.asiste.model.admin.Alerta(
                            codigoOperario = codigoOperario,
                            tipoMarca = com.js.smartvolt.asiste.model.admin.TipoMarca.ENTRADA,
                            hora = registro,
                            faenaId = faena.id,
                            motivo = "Fuera de faena",
                            estado = "REVISION"
                        ))
                        _resultado.value = ResultadoMarca.rechazadaFueraDeFaena(
                            (lectura.distanciaMetros ?: 0f).toInt()
                        )
                    }

                    else -> {
                        val atrasada = registro.isAfter(inicioTurno.plusMinutes(TOLERANCIA_MINUTOS))
                        dao.upsert(
                            nuevaAsistencia(
                                fecha,
                                if (atrasada) EstadoAsistencia.ATRASADA else EstadoAsistencia.CUMPLIDA,
                                inicioTurno, finTurno, registro, lat, lng
                            )
                        )
                        _resultado.value = ResultadoMarca.aprobada(
                            hora = registro.format(FORMATO_HORA),
                            sinConexion = sinConexion,
                            atrasada = atrasada
                        )
                    }
                }
                refrescarHoy(fecha)
            } finally {
                marcando = false
            }
        }
    }

    fun marcarSalida(faena: Faena, fecha: LocalDate, sinConexion: Boolean = false) {
        if (marcando) return
        marcando = true
        viewModelScope.launch {
            try {
                val hoy = dao.getByFecha(fecha).lastOrNull {
                    it.estado == EstadoAsistencia.CUMPLIDA || it.estado == EstadoAsistencia.ATRASADA
                }
                if (hoy == null) {
                    _resultado.value = ResultadoMarca.sinEntrada()
                    return@launch
                }
                if (hoy.registroSalida != null) {
                    _resultado.value = ResultadoMarca.salidaRegistrada(hoy.registroSalida.format(FORMATO_HORA))
                    return@launch
                }

                _presencia.value = _presencia.value.copy(cargando = true, aviso = null)
                val lectura = leerPresencia(faena)
                _presencia.value = lectura

                if (lectura.latitud == null || lectura.longitud == null) {
                    _resultado.value = ResultadoMarca.sinUbicacion(lectura.aviso ?: "No se pudo leer tu ubicación.")
                    return@launch
                }
                if (lectura.simulada) {
                    _resultado.value = ResultadoMarca.rechazadaPorFakeGps(lectura.dentroFaena)
                    return@launch
                }
                if (!lectura.dentroFaena) {
                    _resultado.value = ResultadoMarca.rechazadaFueraDeFaena((lectura.distanciaMetros ?: 0f).toInt())
                    return@launch
                }

                val salida = LocalTime.now().withSecond(0).withNano(0)
                dao.upsert(hoy.copy(registroSalida = salida))
                _resultado.value = ResultadoMarca.salidaRegistrada(salida.format(FORMATO_HORA))
                refrescarHoy(fecha)
            } finally {
                marcando = false
            }
        }
    }

    private suspend fun refrescarHoy(fecha: LocalDate) {
        _asistenciaHoy.value = dao.getByFecha(fecha).lastOrNull()
    }

    private fun nuevaAsistencia(
        fecha: LocalDate,
        estado: EstadoAsistencia,
        inicioTurno: LocalTime,
        finTurno: LocalTime,
        registro: LocalTime,
        lat: Double,
        lng: Double
    ) = Asistencia(
        fecha = fecha,
        estado = estado,
        inicioTurno = inicioTurno,
        finTurno = finTurno,
        registro = registro,
        latitud = lat,
        longitud = lng
    )

    private suspend fun leerPresencia(faena: Faena): PresenciaUi =
        when (val r = ubicacion.obtener()) {
            is ResultadoUbicacion.Ok -> {
                val distancia = FloatArray(1)
                Location.distanceBetween(
                    r.lectura.latitud, r.lectura.longitud,
                    faena.latitud, faena.longitud,
                    distancia
                )
                val dentro = distancia[0] <= faena.radioMetros
                val aviso = if (dentro && r.lectura.precisionMetros > faena.radioMetros) {
                    "Tu GPS tiene baja precisión (±${r.lectura.precisionMetros.toInt()} m). Actívalo en modo preciso."
                } else null

                PresenciaUi(
                    latitud = r.lectura.latitud,
                    longitud = r.lectura.longitud,
                    precisionMetros = r.lectura.precisionMetros,
                    distanciaMetros = distancia[0],
                    dentroFaena = dentro,
                    simulada = r.lectura.simulada,
                    aviso = aviso
                )
            }
            ResultadoUbicacion.SinPermiso ->
                PresenciaUi(aviso = "Necesitas el permiso de ubicación precisa para marcar.")
            ResultadoUbicacion.GpsApagado ->
                PresenciaUi(aviso = "Activa el GPS del teléfono e intenta de nuevo.")
            is ResultadoUbicacion.Error ->
                PresenciaUi(aviso = r.mensaje)
        }
}

class AsistenciaViewModelFactory(
    private val dao: AsistenciaDAO,
    private val ubicacion: UbicacionProvider,
    private val alertaDao: com.js.smartvolt.asiste.model.data.local.AlertaDao? = null
) : ViewModelProvider.Factory {
    @Suppress("UNCHECKED_CAST")
    override fun <T : ViewModel> create(modelClass: Class<T>): T {
        return AsistenciaViewModel(dao, ubicacion, alertaDao) as T
    }
}

data class CheckPresencia(
    val etiqueta: String,
    val valor: String,
    val ok: Boolean
)

data class ResultadoMarca(
    val aprobada: Boolean,
    val titulo: String,
    val subtitulo: String,
    val checks: List<CheckPresencia>,
    val mensaje: String,
    val nota: String? = null,
    val textoBoton: String
) {
    companion object {
        fun aprobada(
            hora: String,
            rssi: Int? = null,
            sinConexion: Boolean = false,
            atrasada: Boolean = false
        ): ResultadoMarca {
            val checks = buildList {
                add(CheckPresencia("GPS dentro de la faena", "Sí", true))
                if (rssi != null) add(CheckPresencia("Baliza BLE de la faena", "$rssi dBm", true))
                add(CheckPresencia("Ubicación simulada", "NO", true))
            }
            return ResultadoMarca(
                aprobada = true,
                titulo = "MARCA REGISTRADA",
                subtitulo = "Entrada registrada a las $hora",
                checks = checks,
                mensaje = if (atrasada) "Tu asistencia quedó guardada con atraso."
                else "Tu asistencia quedó guardada correctamente.",
                nota = when {
                    sinConexion -> "Sin conexión — se guardará en el teléfono"
                    atrasada -> "Marca fuera de la tolerancia del turno"
                    else -> null
                },
                textoBoton = "VOLVER AL INICIO"
            )
        }

        fun rechazadaPorFakeGps(dentroFaena: Boolean): ResultadoMarca = ResultadoMarca(
            aprobada = false,
            titulo = "MARCA RECHAZADA",
            subtitulo = "Se detectó ubicación simulada",
            checks = listOf(
                CheckPresencia("GPS dentro de la faena", if (dentroFaena) "Sí" else "NO", dentroFaena),
                CheckPresencia("Ubicación simulada", "SÍ", false)
            ),
            mensaje = "Tu marca no fue registrada porque el dispositivo reportó ubicación simulada (mock location).",
            nota = "Si no estás usando una app de ubicación falsa, reinicia el GPS y vuelve a intentar.",
            textoBoton = "VOLVER E INTENTAR DE NUEVO"
        )

        fun rechazadaFueraDeFaena(distanciaMetros: Int): ResultadoMarca = ResultadoMarca(
            aprobada = false,
            titulo = "MARCA RECHAZADA",
            subtitulo = "Estás fuera de la faena",
            checks = listOf(
                CheckPresencia("GPS dentro de la faena", "NO", false),
                CheckPresencia("Distancia a la faena", "$distanciaMetros m", false)
            ),
            mensaje = "Tu marca no fue registrada porque estás fuera del radio de la faena.",
            nota = null,
            textoBoton = "VOLVER E INTENTAR DE NUEVO"
        )

        fun sinUbicacion(aviso: String): ResultadoMarca = ResultadoMarca(
            aprobada = false,
            titulo = "MARCA RECHAZADA",
            subtitulo = "No se pudo obtener tu ubicación",
            checks = emptyList(),
            mensaje = aviso,
            nota = null,
            textoBoton = "VOLVER E INTENTAR DE NUEVO"
        )

        fun sinEntrada(): ResultadoMarca = ResultadoMarca(
            aprobada = false,
            titulo = "SALIDA NO REGISTRADA",
            subtitulo = "No tienes entrada registrada hoy",
            checks = emptyList(),
            mensaje = "Debes marcar entrada antes de marcar salida.",
            nota = null,
            textoBoton = "VOLVER"
        )

        fun salidaRegistrada(hora: String): ResultadoMarca = ResultadoMarca(
            aprobada = true,
            titulo = "SALIDA REGISTRADA",
            subtitulo = "Salida registrada a las $hora",
            checks = emptyList(),
            mensaje = "Tu salida fue guardada correctamente.",
            nota = null,
            textoBoton = "VOLVER AL INICIO"
        )
    }
}

sealed interface LoginEstado {
    data object Idle : LoginEstado
    data object Cargando : LoginEstado
    data class Exito(val codigo: String?) : LoginEstado
    data class Error(val mensaje: String) : LoginEstado
}

class LoginViewModel(private val auth: AuthRepository) : ViewModel() {

    private val _estado = MutableStateFlow<LoginEstado>(LoginEstado.Idle)
    val estado: StateFlow<LoginEstado> = _estado.asStateFlow()

    fun login(correo: String, pass: String) {
        val correoLimpio = correo.trim()
        if (correoLimpio.isBlank() || pass.isBlank()) {
            _estado.value = LoginEstado.Error("Completa correo y contraseña")
            return
        }
        if (!android.util.Patterns.EMAIL_ADDRESS.matcher(correoLimpio).matches()) {
            _estado.value = LoginEstado.Error("Correo inválido")
            return
        }
        viewModelScope.launch {
            _estado.value = LoginEstado.Cargando
            try {
                val usuario = auth.login(correo.trim().lowercase(), pass)
                if (usuario != null) {
                    SessionManager.set(usuario)
                    _estado.value = LoginEstado.Exito(usuario.codigo)
                } else {
                    _estado.value = LoginEstado.Error("Correo o contraseña incorrectos")
                }
            } catch (e: Exception) {
                _estado.value = LoginEstado.Error("Error interno: ${e.message}")
            }
        }
    }

    fun limpiarEstado() {
        _estado.value = LoginEstado.Idle
    }
}

class LoginViewModelFactory(private val auth: AuthRepository) : ViewModelProvider.Factory {
    override fun <T : ViewModel> create(modelClass: Class<T>): T {
        @Suppress("UNCHECKED_CAST")
        return LoginViewModel(auth) as T
    }
}

package com.js.smartvolt.asiste.viewmodel

import android.util.Patterns
import androidx.lifecycle.ViewModel
import androidx.lifecycle.ViewModelProvider
import androidx.lifecycle.viewModelScope
import com.js.smartvolt.asiste.model.data.local.UsuarioDAO
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

sealed interface OlvideEstado {
    data object Idle : OlvideEstado
    data object Cargando : OlvideEstado
    data object Enviado : OlvideEstado
    data class Error(val mensaje: String) : OlvideEstado
}

class OlvideClaveViewModel(private val dao: UsuarioDAO) : ViewModel() {

    private val _estado = MutableStateFlow<OlvideEstado>(OlvideEstado.Idle)
    val estado: StateFlow<OlvideEstado> = _estado.asStateFlow()

    fun enviar(correo: String) {
        val limpio = correo.trim().lowercase()

        if (limpio.isBlank()) {
            _estado.value = OlvideEstado.Error("Ingresa tu correo")
            return
        }
        if (!Patterns.EMAIL_ADDRESS.matcher(limpio).matches()) {
            _estado.value = OlvideEstado.Error("Correo inválido")
            return
        }

        viewModelScope.launch {
            _estado.value = OlvideEstado.Cargando

            // En una app real aquí se llama al backend (POST /auth/forgot-password).
            // Con Room solo comprobamos si el correo está registrado.
            dao.existeCorreo(limpio)

            // Mensaje genérico a propósito: no revelamos si el correo existe
            // (evita la enumeración de usuarios).
            _estado.value = OlvideEstado.Enviado
        }
    }

    fun reset() {
        _estado.value = OlvideEstado.Idle
    }
}

class OlvideClaveViewModelFactory(private val dao: UsuarioDAO) : ViewModelProvider.Factory {
    override fun <T : ViewModel> create(modelClass: Class<T>): T {
        @Suppress("UNCHECKED_CAST")
        return OlvideClaveViewModel(dao) as T
    }
}

package com.js.smartvolt.asiste.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.ViewModelProvider
import androidx.lifecycle.viewModelScope
import com.js.smartvolt.asiste.data.repository.FaenaRepository
import com.js.smartvolt.asiste.data.repository.UsuarioRepository
import com.js.smartvolt.asiste.model.admin.Usuario
import com.js.smartvolt.asiste.model.admin.Faena
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class PerfilViewModel(
    private val usuarioRepo: UsuarioRepository,
    private val faenaRepo: FaenaRepository
) : ViewModel() {
    private val _usuario = MutableStateFlow<Usuario?>(null)
    val usuario: StateFlow<Usuario?> = _usuario.asStateFlow()
    private val _faenaNombre = MutableStateFlow<String?>(null)
    val faenaNombre: StateFlow<String?> = _faenaNombre.asStateFlow()

    fun cargar(codigo: String) {
        viewModelScope.launch {
            val u = usuarioRepo.getByCodigo(codigo)
            _usuario.value = u
            if (u != null) {
                faenaRepo.getById(u.faenaId).collect { f ->
                    _faenaNombre.value = f?.nombre
                }
            }
        }
    }
}

class PerfilViewModelFactory(
    private val usuarioRepo: UsuarioRepository,
    private val faenaRepo: FaenaRepository
) : ViewModelProvider.Factory {
    override fun <T : ViewModel> create(modelClass: Class<T>): T {
        @Suppress("UNCHECKED_CAST")
        return PerfilViewModel(usuarioRepo, faenaRepo) as T
    }
}
